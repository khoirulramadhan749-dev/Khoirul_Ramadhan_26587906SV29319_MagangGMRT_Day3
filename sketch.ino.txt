
#include <Wire.h>
#include <ESP32Servo.h>
#define BNO055_ADDR 0x28
#define SDA_PIN 21
#define SCL_PIN 22
#define PIR_PIN 33
Servo servo1, servo2, servo3, servo4, servo5;
bool pirSebelumnya = false;
bool gerakanPIR = false;
unsigned long waktuMulaiPIR = 0;
float yawSebelumnya = 0;
unsigned long waktuYawBerubah = 0;
int bacaRegister(uint8_t reg) {
  Wire.beginTransmission(BNO055_ADDR);
  Wire.write(reg);
  if (Wire.endTransmission(false) != 0) return 0;
  Wire.requestFrom(BNO055_ADDR, (uint8_t)2);
  if (Wire.available() < 2) return 0;
  uint8_t lowByte = Wire.read();
  uint8_t highByte = Wire.read();
  return (int16_t)((highByte << 8) | lowByte);
}
float bacaSudut(uint8_t reg) {
  return bacaRegister(reg) / 16.0;
}
int batasServo(float sudut) {
  return constrain((int)(90 + sudut), 0, 180);
}
int hitungPWM(int sudut) {
  return map(sudut, 0, 180, 1000, 2000);
}
void setup() {
  Serial.begin(115200);
  Wire.begin(SDA_PIN, SCL_PIN);
  servo1.attach(13);
  servo2.attach(14);
  servo3.attach(25);
  servo4.attach(26);
  servo5.attach(27);
  servo1.write(90);
  servo2.write(90);
  servo3.write(90);
  servo4.write(90);
  servo5.write(90);
  pinMode(PIR_PIN, INPUT);
  waktuYawBerubah = millis();
  Serial.println("Sistem siap");
}
void loop() {
  float yaw = bacaSudut(0x1A);
  float roll = bacaSudut(0x1C);
  float pitch = bacaSudut(0x1E);
  int kondisiPIR = digitalRead(PIR_PIN);
  if (kondisiPIR == HIGH && !pirSebelumnya && !gerakanPIR) {
    gerakanPIR = true;
    waktuMulaiPIR = millis();
    Serial.println("Gerakan terdeteksi");
  }
  if (gerakanPIR) {
    servo1.write(120);
    servo2.write(60);
    servo3.write(120);
    servo4.write(60);
    servo5.write(120);
    if (millis() - waktuMulaiPIR >= 1000) {
      gerakanPIR = false;
      Serial.println("PIR selesai");
    }
    pirSebelumnya = (kondisiPIR == HIGH);
    return;
  }
  pirSebelumnya = (kondisiPIR == HIGH);
  int posisi1 = batasServo(-roll);
  int posisi2 = batasServo(roll);
  int posisi3 = batasServo(pitch);
  int posisi4 = batasServo(pitch);
  servo1.write(posisi1);
  servo2.write(posisi2);
  servo3.write(posisi3);
  servo4.write(posisi4);
  if (abs(yaw - yawSebelumnya) > 0.5) {
    waktuYawBerubah = millis();
    yawSebelumnya = yaw;
    servo5.write(batasServo(yaw));
  }
  if (millis() - waktuYawBerubah >= 1000) {
    servo5.write(90);
  }
  static unsigned long waktuCetak = 0;
  if (millis() - waktuCetak >= 500) {
    waktuCetak = millis();
    Serial.print("Roll: ");
    Serial.print(roll);
    Serial.print(" | Pitch: ");
    Serial.print(pitch);
    Serial.print(" | Yaw: ");
    Serial.println(yaw);
    Serial.print("PWM Servo 1: ");
    Serial.print(hitungPWM(posisi1));
    Serial.print(" | Servo 2: ");
    Serial.print(hitungPWM(posisi2));
    Serial.print(" | Servo 3: ");
    Serial.print(hitungPWM(posisi3));
    Serial.print(" | Servo 4: ");
    Serial.println(hitungPWM(posisi4));
  }
}
